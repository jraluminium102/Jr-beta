3.2 พฤติกรรม

  • กดชิป → set .i-prod value = prodId ตัวแทน → dispatchEvent('change') (trigger buildItemOpts/calcQuote เดิมครบ ไม่ต้องแก้)
  • dropdown เดิม ยังอยู่ แต่ relabel เป็น "รุ่น" — user เปลี่ยนรุ่นย่อยในชนิดเดิมได้ (เช่น sliding_euro → sliding_sms)
  • Nice-to-have: กรอง option ใน dropdown ให้เหลือเฉพาะ optgroup ของชนิดที่เลือก + ปุ่ม "แสดงทั้งหมด" (ลดความยาว) · ถ้าไม่ทำ ให้ dropdown โชว์เต็มได้
  • ตอนเปิดการ์ดที่มี .i-prod ค่าอยู่แล้ว → highlight ชิปชนิดที่ตรงกับ cat ของ prod ปัจจุบัน (อ่าน PBYID[val].cat map กลับเป็นชิป)

3.3 Pattern (อิง references/chip-pattern.md)

// ใน refreshSet() สาขา isGH — ต่อจากที่ groupGHOpts(c) ทำงาน
// inject ครั้งเดียวต่อการ์ด (เช็ค flag กัน duplicate เหมือน gh-opt-cat-det)
function injectTypeChips(ch){
  var prodSel = ch.querySelector('.i-prod'); if(!prodSel) return;
  var wrap = prodSel.closest('.full'); if(!wrap || wrap.querySelector('.gh-type-chips')) return;
  var MAP = [
    {t:'บานเลื่อน', id:'sliding_euro'}, {t:'บานเปิด', id:'casement_euro'},
    {t:'บานเฟี้ยม', id:'folding_euro'}, {t:'กระจกติดตาย', id:'fixed_glass'},
    {t:'ระแนงบังตา', id:'/*default ระแนงบังตา*/'}, {t:'บานเปลือย', id:'frameless_fixed'}
  ];
  var bar = document.createElement('div'); bar.className='gh-type-chips';
  bar.innerHTML = MAP.map(function(m){
    return '<button type="button" class="gh-tchip" data-id="'+m.id+'">'+m.t+'</button>';
  }).join('') + '<button type="button" class="gh-tchip more" data-id="">อื่นๆ ▾</button>';
  wrap.insertBefore(bar, prodSel);           // ชิปอยู่เหนือ dropdown "รุ่น"
  bar.querySelectorAll('.gh-tchip').forEach(function(b){
    b.addEventListener('click', function(){
      bar.querySelectorAll('.gh-tchip').forEach(function(x){x.classList.remove('on');});
      b.classList.add('on');
      if(b.dataset.id){ prodSel.value=b.dataset.id; prodSel.dispatchEvent(new Event('change',{bubbles:true})); }
      // (option) filter dropdown ตาม optgroup ของชนิดที่เลือก
    });
  });
  // relabel dropdown เดิม
  var lbl = wrap.querySelector('label'); if(lbl) lbl.textContent='รุ่น (ปรับได้)';
}

CSS ชิป: min-height:44px; border:1.5px solid; border-radius:11px; · on = พื้นแดง #B3151D ขาว (ตาม chip-pattern.md)

4. SHOULD stepper "จำนวนบาน" .i-panels

  • ทำใน addItem template (L2219) — มีผลทั้ง G1+G6 (stepper ดีกับทุกกลุ่ม ความเสี่ยงต่ำ)
  • ครอบ <input class="i-panels"> ด้วยปุ่ม − / + : กดแล้ว value = max(1, value±1) → fire input (มี listener sync mirror อยู่แล้ว L2269)
  • ห้ามลบ <input class="i-panels"> — ยังพิมพ์เลขเองได้ (ปุ่มแค่เสริม) · listener เดิมที่ผูก i-panels ต้องทำงานเหมือนเดิม

5. เคาะก่อน ชิป o-opentype / o-bottomrail (กระทบกลุ่ม 1)

.o-opentype (5) และ .o-bottomrail (2) สร้างใน buildItemOpts() (L2346) — ใช้ร่วมกับกลุ่ม 1
ถ้าทำชิปใน buildItemOpts → กลุ่ม 1 ได้ชิปด้วย (สม่ำเสมอดี แต่ G1 form-ux ผ่านแล้ว ต้องเทสซ้ำ)
ต้องให้พี่นัทเคาะ: (ก) ทำชิปทั้ง G1+G6 · (ข) เฉพาะ G6 (ห่อใน refreshSet GH) · (ค) ข้ามไปก่อน
