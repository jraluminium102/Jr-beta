1. ขอบเขตงาน

เปลี่ยนแค่ "วิธีกรอก" (control type) — ไม่แตะ logic/ราคา/ค่าที่ส่งเข้าใบ
ชิปและ stepper ต้อง sync ค่ากลับเข้า element เดิมที่ readItem() อ่าน (class .o-* / .i-* เดิม) — ห้ามเปลี่ยน value, class, หรือลำดับ option ใดๆ
กฎเหล็ก: readItem() อ่านค่าจาก d.querySelector('.i-type').value, d.querySelector('.o-bottomrail').value ฯลฯ — element เหล่านี้ต้องยังคงอยู่ใน DOM (hidden input หรือ select ที่ซ่อน) พร้อมค่าที่ถูกต้องเสมอ ก่อนที่ calcQuote() จะรัน

สรุปงานทั้งหมด

  • 1️⃣ ย้าย .i-prod (แบบ) ขึ้นมาเป็นลำดับ 2 (ทันทีหลัง .i-group)
  • 2️⃣ เปลี่ยน .i-type (ประเภท) จาก select → ชิป 2 ปุ่มใหญ่ (52px)
  • 3️⃣ เปลี่ยน select ≤4 ตัวเลือก → ชิป (8 ช่อง รายละเอียดในตาราง)
  • 4️⃣ ก้อนใหม่: รวบมือจับ → 1 กล่อง multi-toggle (ติ๊กได้หลายแบรนด์พร้อมกัน) พร้อม X-series (ฟรี) + Cmech หลบมุ้ง
  • 5️⃣ ก้อนใหม่: รวบมุ้ง 4 ช่องกระจาย → 1 กล่อง (โผล่/ซ่อนตาม o-mosq)
  • 6️⃣ ปุ่ม/ชิปทุกตัวสูง ≥40px (tap target มือถือ)

3A. Consolidate ก้อน 1 — มือจับ (multi-toggle)

มติพี่นัท 2026-06-11 (อัปเดต): ใส่มือจับได้หลายอย่างพร้อมกันในบานเดียว (เช่น ดิจิตอลล็อค + มือจับดึงสแตนอร่าม)
pattern เปลี่ยนจาก "เลือก 1 แบรนด์ (clear อื่น)" → multi-toggle อิสระ
ติ๊กแบรนด์ไหน = โผล่ sub นั้น · ปิดแบรนด์ = เคลียร์ค่า + ซ่อน sub · บวกราคาสะสมทุกแบรนด์ที่ติ๊ก
สถานะปัจจุบัน (index.html L2281–2299): มือจับ 4 ช่องโผล่พร้อมกันเสมอถ้า p.digihandle === true — เซลล์ไม่รู้ว่าเลือกหลายแบรนด์พร้อมกันได้หรือไม่ · แต่ละช่องว่างก็เสี่ยงคิดราคาผิด

การทำงานที่ต้องการ (multi-toggle)

  1. แสดง toggle แบรนด์ 5 ตัว: Cmech · ดิจิตอล · สแตนอร่าม · X-series (ฟรี) · รุ่นอื่น ✎
  2. แต่ละตัวเป็น toggle อิสระ — กดครั้งแรก = ติ๊ก (โผล่ sub) · กดซ้ำ = ปิด (เคลียร์ค่า + ซ่อน sub)
  3. ห้ามเคลียร์แบรนด์อื่นเมื่อติ๊กตัวใหม่ — แต่ละ sub/control ทำงานอิสระ
  4. ยังไม่ติ๊กอะไร = ซ่อน sub ทุกตัว (ฟอร์มสั้น เหมือนเดิม)
  5. readItem / calcQuote อ่านค่าจาก .o-cmech / .o-digi / .o-stainless / .o-handlename / .o-handleprice ตามปกติ — บวกราคาสะสมทุกที่ที่มีค่า

JS function: toggleHandleBrand + cmechTypePick

// เพิ่มใน script global (แทน handleBrandPick เดิม)
// multi-toggle อิสระ — ไม่ clear แบรนด์อื่น
function toggleHandleBrand(btn) {
  const brand = btn.dataset.hbrand;
  const isOn = btn.classList.contains('on');

  if (isOn) {
    // ปิด: เคลียร์ค่า control ของแบรนด์นี้เท่านั้น + ซ่อน sub
    btn.classList.remove('on');
    clearHandleBrandControls(btn, brand);
    const sub = btn.closest('.handle-brand-wrap').querySelector('.hsub-' + brand);
    if (sub) sub.style.display = 'none';
  } else {
    // เปิด: ติ๊กแบรนด์ + โผล่ sub + set default
    btn.classList.add('on');
    const sub = btn.closest('.handle-brand-wrap').querySelector('.hsub-' + brand);
    if (sub) sub.style.display = '';
    // default Cmech = embed
    if (brand === 'cmech') {
      const ch = btn.closest('.ch');
      if (ch) { const s = ch.querySelector('.o-cmech'); if (s) s.value = 'embed'; }
    }
    // X-series: set flag
    if (brand === 'xseries') {
      const ch = btn.closest('.ch');
      if (ch) { const xh = ch.querySelector('.o-xhandle'); if (xh) xh.value = '1'; }
    }
  }
  calcQuote();
}

// เคลียร์ค่า control เฉพาะแบรนด์ที่กำหนด (ไม่แตะแบรนด์อื่น)
function clearHandleBrandControls(btn, brand) {
  const ch = btn.closest('.ch');
  if (!ch) return;
  if (brand === 'cmech') {
    const s = ch.querySelector('.o-cmech'); if (s) s.value = '';
    const a = ch.querySelector('.o-cmechawn'); if (a) a.checked = false;
  } else if (brand === 'digital') {
    const s = ch.querySelector('.o-digi'); if (s) s.value = '';
  } else if (brand === 'stainless') {
    const s = ch.querySelector('.o-stainless'); if (s) s.value = '';
  } else if (brand === 'xseries') {
    const xh = ch.querySelector('.o-xhandle'); if (xh) xh.value = '';
  } else if (brand === 'other') {
    const hn = ch.querySelector('.o-handlename'); if (hn) hn.value = '';
    const hp = ch.querySelector('.o-handleprice'); if (hp) hp.value = '0';
  }
}

// sync ชิป Cmech type → select ซ่อน (ไม่เปลี่ยน)
function cmechTypePick(btn) {
  const sub = btn.closest('.hsub-cmech');
  if (!sub) return;
  sub.querySelectorAll('[data-val]').forEach(function(c) { c.classList.remove('on'); });
  btn.classList.add('on');
  const sel = sub.querySelector('.o-cmech');
  if (sel) { sel.value = btn.dataset.val; sel.dispatchEvent(new Event('change', {bubbles:true})); }
}

3B. Consolidate ก้อน 2 — มุ้ง (4 ช่อง → 1 กล่อง)

สถานะปัจจุบัน (index.html L2461–2469): ช่อง o-mosqw, o-mosqh, o-mosqpanels, o-mosqfabric, o-mosqcolor โผล่ทั้งหมด แม้ยังเลือก o-mosq = "" (ไม่ใส่) — เซลล์งงว่าต้องกรอกหรือเปล่า
เป้าหมาย: ซ่อนทั้งหมดถ้า o-mosq = "" · โผล่เมื่อเลือกมุ้ง · จัดอยู่กล่องเดียวกัน

การทำงานที่ต้องการ

  1. dropdown .o-mosq อยู่บนสุดกล่อง — ค่า default = "" (ไม่ใส่)
  2. ถ้า .o-mosq = "" → ซ่อน: ผ้ามุ้ง + ขนาด + จำนวนบาน + สีกรอบทั้งหมด
  3. ถ้า .o-mosq มีค่า → โผล่ทั้งกลุ่มในกล่องสีฟ้าใต้ dropdown
  4. สีกรอบมุ้ง (.o-mosqcolor) เปลี่ยนจาก dropdown → ชิป 4 ตัว (ตามสีบาน / ซาฮาร่า / KL พิเศษ / KL ลายไม้)
  5. ผ้ามุ้ง (.o-mosqfabric) คง dropdown (5 ตัว)
  6. ขนาดมุ้ง (o-mosqw/h) เว้นว่าง = auto จากบานหลัก — label "auto จากบาน" · ถูกต้องแล้ว คงไว้

Helper function: chSetChip (เพิ่มใน JS global)

function chSetChip(btn, selectClass) {
  const ch = btn.closest('.ch');
  if (!ch) return;
  const val = btn.dataset.val;
  ch.querySelectorAll(`.chip[onclick*="${selectClass}"]`).forEach(function(c) {
    c.classList.remove('on');
  });
  btn.classList.add('on');
  const sel = ch.querySelector('.' + selectClass);
  if (sel) {
    sel.value = val;
    sel.dispatchEvent(new Event('change', { bubbles: true }));
  }
}
