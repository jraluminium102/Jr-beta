function mosqFabricColors(fabBtn){
  var ch=fabBtn.closest('.ch'); var fab=fabBtn.dataset.val;
  var MAP={fiber:['ดำ','เทา'],cat:['ขาว'],rat:['เงิน','ดำ'],safety:['ดำ']};
  var cols=MAP[fab]||['ดำ']; var fixed=cols.length===1;
  var wrap=ch.querySelector('.o-sfc-chipwrap'); var sel=ch.querySelector('.o-screenfabriccolor');
  wrap.innerHTML=cols.map(function(c,i){return fixed
    ? '<button type="button" class="chip on" disabled>'+c+'</button>'
    : '<button type="button" class="chip'+(i===0?' on':'')+'" data-val="'+c+'" onclick="chSetChip(this,\'o-screenfabriccolor\')">'+c+'</button>';}).join('')
    + (fixed?'<span class="opt" style="font-size:11px;color:#0369A1;">สีเดียว</span>':'');
  if(sel){ sel.value=cols[0]; sel.dispatchEvent(new Event('change',{bubbles:true})); }
}

3) สูตรราคา — ยืนยันแล้ว ไม่ต้องแก้ (cross-check 9/11 ตรง engine)

screen_addon (imp21/22/23): sell = ปัดพัน( max( A×(rate+ค่าผ้า) , min×จำนวนบาน ) )
A = พื้นที่รวม (ไม่คูณบาน) · min×panels · ค่าผ้า/ตร.ม.: fiber 0, กันแมว 800, กันหนู 1,200, นิรภัย 3,500 (บวกเข้า base ก่อนเทียบ min)
non-addon: sell = ปัดพัน( max( min, A×rate ) )
ทุกราคาปัดขึ้นพัน (roundUp = Math.ceil(x/1000)*1000)

⚠ เช็คซ้ำ: เคส "หน้าต่าง + min ชนะ" — calcUnit อ่าน itype จาก DOM .i-type (ตัวแปร d บรรทัด ~1056) ไม่ใช่ optSel · ใช้งานจริงในเบราว์เซอร์ window→min_window ถูก · ยืนยันในเบราว์เซอร์ตอนทำ
