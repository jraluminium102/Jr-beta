Pattern JS — ฟังก์ชัน 3 ตัว

/* ตาราง: กลุ่ม → array of {id, name} */
var RF_PRODS = {
  metal:  [{id:'imp7',n:'1" PVC EPS'},{id:'imp8',n:'2" PVC EPS'},
            {id:'imp9',n:'1" เหล็ก EPS'},{id:'imp10',n:'2" เหล็ก EPS'},
            {id:'imp11',n:'1" ฟอยล์ PU'},{id:'imp12',n:'2" ฟอยล์ PU'},
            {id:'imp13',n:'1" เหล็ก PU'},{id:'imp14',n:'2" เหล็ก PU'}],
  chinco: [{id:'imp15',n:'6มม. Heat Cut'},{id:'imp16',n:'6มม. Superior'},
            {id:'imp17',n:'6มม. Nature'},{id:'imp18',n:'4มม. Shade'},
            {id:'imp19',n:'10มม. Prime'},{id:'imp20',n:'10มม. Grand'}],
  poly:   [{id:'roof_polyton',n:'โพลีตัน 3มม.'},{id:'roof_laminate',n:'กระจกลามิเนต'}],
  vinyl:  [{id:'roof_vinyl',n:'ไวนิล VG'},{id:'roof_delight',n:'ดีไลท์'}],
};

/* กดชิปกลุ่ม */
function rfPickGrp(btn){
  var ch = btn.closest('.ch');
  ch.querySelectorAll('#rf-grp-chips .chip').forEach(function(c){c.classList.remove('on');});
  btn.classList.add('on');
  rfBuildProdChips(ch, btn.dataset.grp);
}

/* inject ชิปรุ่น + set default = ตัวแรก */
function rfBuildProdChips(ch, grp){
  var box = ch.querySelector('#rf-prod-chips'); if(!box) return;
  var prods = RF_PRODS[grp] || [];
  box.innerHTML = prods.map(function(p,i){
    return '';
  }).join('');
  /* set default */
  if(prods.length) rfSyncProd(ch, prods[0].id);
}

/* กดชิปรุ่น → sync .i-prod + fire change (⚠ fire change ต้องทำเสมอ) */
function rfPickProd(btn){
  var ch = btn.closest('.ch');
  ch.querySelectorAll('#rf-prod-chips .chip').forEach(function(c){c.classList.remove('on');});
  btn.classList.add('on');
  rfSyncProd(ch, btn.dataset.id);
}

/* sync .i-prod + dispatchEvent — DEV: นี่คือจุดเชื่อม engine */
function rfSyncProd(ch, prodId){
  var sel = ch.querySelector('.i-prod'); if(!sel) return;
  /* update option หรือ inject ถ้าขาด */
  if(!sel.querySelector('option[value="'+prodId+'"]')){
    var o = document.createElement('option');
    o.value = prodId; o.textContent = prodId;
    sel.appendChild(o);
  }
  sel.value = prodId;
  /* ⚠ fire change → buildItemOpts rebuild ออปชั่น */
  sel.dispatchEvent(new Event('change',{bubbles:true}));
}
จุดระวัง — .i-prod sync:
1. dispatchEvent('change') ต้องทำทุกครั้งที่กดชิปรุ่น — ไม่อย่างนั้น groupGHOpts ไม่ rebuild · ราคาไม่ update
2. rfBuildProdChips ต้อง call ทุกครั้งที่ buildItemOpts สร้าง form ใหม่ (เช่น เพิ่มชุดใหม่) — ไม่ใช่แค่ครั้งแรก
3. ถ้า .i-prod มี option imp7–imp20/roof_* อยู่แล้ว ไม่ต้อง inject ซ้ำ — เช็ค querySelector('option[value="..."]') ก่อน
4. groupGHOpts ใช้ isRoof = !!(ch.querySelector('.o-roofend') || ch.querySelector('.o-roofbatten')) เพื่อตรวจ — ไม่ต้องแก้

สถานะปัจจุบัน (บรรทัด ~2714 — แสดงตลอด):

// ก่อนแก้ (index.html ~2714)
h+='<label class="opt">ปิดซ่อนราง: กว้างราง(ซม.) '

h+='<div class="o-rfgc-wrap">'
  +'<label class="opt">'

  +' onchange="var r=this.closest(\'.ch\').querySelector(\'.o-rfgc-reveal\');'
  +'if(r)r.style.display=this.checked?\'flex\':\'none\'; calcQuote();"> '
  +'มีงานปิดซ่อนราง'
  +'</label>'
  +'<span class="o-rfgc-reveal" style="display:none;flex-wrap:wrap;gap:4px 10px;">'
  +' กว้างราง(ซม.) <input type="number" class="o-rfgcw" min="0" step="1" value="0"'
  +' style="width:55px;height:30px;" oninput="calcQuote();">'
  +' สูง(ซม.) <input type="number" class="o-rfgch" min="0" step="1" value="10"'

6. DoD (Definition of Done)

  • กดชิปกลุ่มวัสดุ (เมทัลชีท → ชินโคไลท์ → โพลีตัน/กระจก → ไวนิล/ดีไลท์) → ชิปรุ่นในกลุ่มโหลดใหม่ถูกต้อง
  • กดชิปรุ่น → .i-prod.value เปลี่ยน + buildItemOpts rebuild ออปชั่นหลังคา (เช่น ชิปสีเปลี่ยน)
  • ราคาหลังคาไม่เพี้ยน — เทสด้วยชุดที่มีอยู่: กดรุ่นต่างๆ ราคาเปลี่ยนตาม bucket เดิม
  • ปิดซ่อนราง: ก่อนติ๊ก = ซ่อน 3 ช่อง · หลังติ๊ก = reveal · ถอด = ซ่อนอีกครั้ง
  • ครอบคาน: ก่อนติ๊ก = ซ่อน 2 ช่อง · หลังติ๊ก = reveal · ถอด = ซ่อนอีกครั้ง
  • ของเสริม (ปิดซ่อนราง/ครอบคาน) เมื่อ reveal+กรอก → ราคาคำนวณรวมถูกต้อง (ไม่เป็น 0)
  • เพิ่มชุดหลังคาใหม่ (คลิก +เพิ่มชุด) → ชิปกลุ่ม/รุ่นโหลดขึ้นมาพร้อมใช้ (ไม่ใช่ว่างเปล่า)
  • ออปชั่นอื่นใน G3 (ปลายหลังคา/รางน้ำ/เลื่อน/ของเสริมอื่นๆ) ยังคำนวณครบ · ไม่ regression
  • quote-fidelity PASS: ใบเสนอราคา G3 ที่สร้างจาก index.html มี product name/ออปชั่น ตรงกับที่กรอก
