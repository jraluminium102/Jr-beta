// ══ ธงC: ceilCalc — คำนวณราคาฝ้า addon (ใช้ทั้งใน calcQuote และ display) ══
// anchor: ค้นหา "function ceilCalc" หรือวางก่อน "function calcQuote"
function ceilCalc(ceilTypeId, ceilAreaSqm, ceilColor, smoothSlope, smoothInsul){
  var p=PBYID[ceilTypeId]; if(!p) return {sell:0, msg:''};
  var A=Math.max(ceilAreaSqm||0, 0);
  if(p.ceiling){ // ceiling_smooth
    var effA=Math.max(A,10);
    var rate=10000; // fallback
    if(RATES.CEILING){ rate=rateOf(effA,RATES.CEILING); }
    var base=effA*rate;
    var insulUp=(smoothInsul?600*effA:0);
    var sell=roundUp(base+insulUp);
    var msg='ฝ้าฉาบเรียบ '+effA.toFixed(1)+' ตร.ม. × '+fmt(rate)+(A<10?' (min 10 ตร.ม.)':'')+(insulUp>0?' + ฉนวน +'+fmt(insulUp):'');
    return {sell:sell, msg:msg};
  }
  if(p.crates){ // ceil_cshape / ceil_bsc
    var ck=ceilColor&&p.crates[ceilColor]!==undefined?ceilColor:Object.keys(p.crates)[0];
    var rate2=p.crates[ck];
    var sell2=roundUp(A*rate2);
    var msg2=p.name.replace(/^ฝ้า\s*/,'')+' '+A.toFixed(1)+' ตร.ม. × '+fmt(rate2)+'/ตร.ม.'+' ('+ck+')';
    return {sell:sell2, msg:msg2};
  }
  return {sell:0, msg:''};
}
// ══ จบ ceilCalc ══
