function _noteBlocks(note){
  if(!note) return '';
  var O=[],N=[];
  String(note).split(/\n+/).forEach(function(ln){ ln=(ln||'').trim(); if(!ln)return;
    if(ln.indexOf('OPTION')===0) O.push(ln.replace(/^OPTION\s*[:：]?\s*/,''));
    else N.push(ln.replace(/^หมายเหตุ\s*[:：]?\s*/,'')); });
  var h='';
  if(O.length){ h+='<div style="margin-top:8px;font-weight:600;color:#B3151D;">OPTION</div>';
    O.forEach(function(o){ h+='<div style="margin:1px 0 0 8px;color:#B3151D;font-size:13px;">- '+o+'</div>'; }); }
  if(N.length){ h+='<div style="margin-top:8px;font-weight:600;color:#374151;">หมายเหตุ</div>';
    N.forEach(function(n){ h+='<div style="margin:1px 0 0 8px;color:#6B7280;font-size:13px;">- '+n+'</div>'; }); }
  return h;
}
