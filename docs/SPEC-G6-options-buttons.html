7. Pattern — ชิป single-select (ใช้ทั้ง G1+G6)

// chSetChip(btn, selectClass) — มีอยู่แล้วใน index.html (global helper)
// ใช้ได้เลย ไม่ต้องเพิ่ม
function chSetChip(b, cls){
  var ch = b.closest('.ch');
  var sel = ch ? ch.querySelector('.'+cls) : null;
  if(!sel) return;
  b.closest('.chip-group').querySelectorAll('.chip').forEach(function(c){ c.classList.remove('on'); });
  b.classList.add('on');
  sel.value = b.dataset.val;
  sel.dispatchEvent(new Event('change', {bubbles:true}));
}

10. จุดแก้ใน groupGHOpts — หมวด "ครอบวงกบ" เข้า "อื่นๆ"

ปัจจุบัน: keyword ใน CATS (non-roof) เป็น keyword-based — ตรวจ textContent+className ของ child
ปัญหา: .o-fcsides label text = "ครอบวงกบอลู" → match หมวด "🚪 ชนิด/การเปิด" (kw มี 'ราวจับ' แต่ไม่มี 'ครอบวงกบ'?) ต้องตรวจ render จริง
แก้: เพิ่ม 'ครอบวงกบ' เข้า kw หมวด "➕ อื่นๆ" หรือตัดออกจาก kw หมวดอื่น ตรวจสอบก่อน 
// CATS non-roof ปัจจุบัน — ตรวจ keyword ที่อาจจับ "ครอบวงกบ" ผิดหมวด:
// หมวด 🚪: kw มี 'ราวจับ' → ไม่ match ครอบวงกบ (OK)
// หมวด 🔧: kw ไม่มี 'ครอบ' → ไม่ match (OK)
// หมวด ➕ (last bucket): ทุกที่ที่ไม่ match → ตกหมวด "อื่นๆ" อัตโนมัติ
// ผลคือ fcsides/solidlower/uchannel/extrawork/removeold/dfm ตกหมวด "อื่นๆ" แล้ว (ถูกต้อง)
