BLOCK 1 — cascade ระแนง (แทนทั้งบล็อก: ตั้งแต่ function rnCatKey ถึงจบ function rnDoorSwitch)

// ===== ระแนง cascade ใหม่: กล่อง → หน้าโชว์ → ระยะห่าง → โครง (รวม บังตา+ผนัง) =====
function rnCatKey(p){ return (p.cat==='ระแนง-บังตา'||p.cat==='ระแนง-ผนัง')?'box':(p.cat==='ระแนง-เกล็ด'?'gleed':null); }
function rnParse(p){ // → {box,face,gap,frame}(กล่อง) | {size,frame}(เกล็ด)
  var nm=p.name||'', frame=/รวมโครง/.test(nm)?'รวมโครง':'ไม่โครง';
  if(p.cat==='ระแนง-เกล็ด'){ return {size:nm.replace(/^เกล็ด\s*Z\s*/,'').replace(/(ไม่|รวม)โครง\s*$/,'').trim(), frame:frame}; }
  var body=nm.replace(/^บังตาผนังระแนง/,'').replace(/^ระแนงผนัง\s*/,'').replace(/(ไม่|รวม)โครง\s*$/,'').trim();
  var parts=body.split(/โชว์/); var box=(parts[0]||'').trim(), sp=(parts[1]||'').trim();
  var fp=sp.split(/\s+/); var face=fp[0]||'', gap=fp.slice(1).join(' ')||'ทึบ';
  return {box:box, face:face, gap:gap, frame:frame};
}
var RN_TREE={box:{}, gleed:{}}, RN_BOX_ORDER=[];
PRODUCTS.forEach(function(p){ var ck=rnCatKey(p); if(!ck)return; var d=rnParse(p);
  if(ck==='box'){
    if(!RN_TREE.box[d.box]){ RN_TREE.box[d.box]={}; RN_BOX_ORDER.push(d.box); }
    (RN_TREE.box[d.box][d.face]=RN_TREE.box[d.box][d.face]||{});
    (RN_TREE.box[d.box][d.face][d.gap]=RN_TREE.box[d.box][d.face][d.gap]||{})[d.frame]=p.id;
  } else { (RN_TREE.gleed[d.size]=RN_TREE.gleed[d.size]||{})[d.frame]=p.id; }
});
function rnRate(id){ var p=PBYID[id]; return p?p.rate:0; }
function rnResolve(ck,k1,k2,k3,frame){ // box: k1=กล่อง k2=หน้าโชว์ k3=ระยะ · gleed: k1=ขนาด
  if(ck==='gleed'){ var g=RN_TREE.gleed[k1]; if(!g)return''; return g[frame]||g['ไม่โครง']||g[Object.keys(g)[0]]||''; }
  var b=RN_TREE.box[k1]; if(!b)return''; if(!b[k2])k2=Object.keys(b)[0]; var fc=b[k2]; if(!fc)return'';
  if(!fc[k3])k3=Object.keys(fc)[0]; var node=fc[k3]; if(!node)return'';
  return node[frame]||node['รวมโครง']||node['ไม่โครง']||node[Object.keys(node)[0]]||'';
}
function rnPick(btn,dim){
  var ch=btn.closest('.ch'); if(!ch)return; var ps=ch.querySelector('.i-prod'); var p=ps?PBYID[ps.value]:null; if(!p)return;
  var ck=rnCatKey(p); if(!ck)return; var d=rnParse(p), v=btn.dataset.val;
  var k1,k2,k3,frame;
  if(ck==='box'){
    k1=d.box; k2=d.face; k3=d.gap; frame=d.frame;
    if(dim==='box'){ k1=v; var b=RN_TREE.box[v]; k2=Object.keys(b)[0]; k3=Object.keys(b[k2])[0]; frame='รวมโครง'; ch.dataset.rnStep='2'; }   // เลือกกล่อง → เปิดหน้าโชว์ · ★default รวมโครง
    else if(dim==='face'){ k2=v; k3=Object.keys(RN_TREE.box[k1][v])[0]; frame='รวมโครง'; ch.dataset.rnStep='3'; }  // เลือกหน้าโชว์ → เปิดระยะ
    else if(dim==='gap'){ k3=v; frame='รวมโครง'; ch.dataset.rnStep='4'; }  // เลือกระยะ → เปิดโครง
    else if(dim==='frame'){ frame=v; }
  } else {
    k1=d.size; frame=d.frame;
    if(dim==='k1'){ k1=v; frame='รวมโครง'; ch.dataset.rnStep='2'; }
    else if(dim==='frame'){ frame=v; }
  }
  var id=rnResolve(ck,k1,k2,k3,frame); var cur=ch.querySelector('.i-prod').value;
  if(id && id!==cur) g3SyncProd(ch,id);
  else { buildItemOpts(ch); calcQuote(); }
}
function _rnStepHint(){ return '<div style="font-size:11px;color:#9CA3AF;margin-top:6px;">↓ เลือกด้านบนก่อน แล้วตัวเลือกถัดไปจะโผล่</div>'; }
function rnSelectorHTML(p, ch){
  var ck=rnCatKey(p); if(!ck)return''; var d=rnParse(p);
  var step=parseInt((ch&&ch.dataset.rnStep)||'1')||1;
  var h='<div class="opt rn-prodsel" style="width:100%;border:1px solid #E5E7EB;border-radius:8px;padding:8px 10px;background:#FAFAFA;">';
  var more=false;
  if(ck==='box'){
    h+='<label class="opt" style="width:100%;margin:0;"><b style="color:#B3151D;">①</b> ขนาดกล่อง '+rnChips(RN_BOX_ORDER.filter(function(b){return RN_TREE.box[b];}),d.box,'box')+'</label>';
    if(step>=2){ var faces=RN_TREE.box[d.box]?Object.keys(RN_TREE.box[d.box]):[];
      h+='<label class="opt" style="width:100%;margin:8px 0 0;"><b style="color:#B3151D;">②</b> หน้าโชว์ '+(faces.length>1?'<span style="font-size:10px;color:#0369A1;">('+faces.length+' หน้า)</span> ':'')+rnChips(faces,d.face,'face')+'</label>';
    } else more=true;
    if(step>=3){ var gaps=(RN_TREE.box[d.box]&&RN_TREE.box[d.box][d.face])?Object.keys(RN_TREE.box[d.box][d.face]):[];
      h+='<label class="opt" style="width:100%;margin:8px 0 0;"><b style="color:#B3151D;">③</b> ระยะห่าง '+rnChips(gaps,d.gap,'gap')+'</label>';
    } else if(step>=2) more=true;
    if(step>=4){ var frs=(RN_TREE.box[d.box]&&RN_TREE.box[d.box][d.face]&&RN_TREE.box[d.box][d.face][d.gap])?Object.keys(RN_TREE.box[d.box][d.face][d.gap]):['ไม่โครง','รวมโครง'];
      h+='<label class="opt" style="width:100%;margin:8px 0 0;"><b style="color:#B3151D;">④</b> โครง '+rnChips(frs,d.frame,'frame')+'</label>';
    } else if(step>=3) more=true;
  } else {
    h+='<label class="opt" style="width:100%;margin:0;"><b style="color:#B3151D;">①</b> ขนาดเกล็ด '+rnChips(Object.keys(RN_TREE.gleed),d.size,'k1')+'</label>';
    if(step>=2){ var frg=RN_TREE.gleed[d.size]?Object.keys(RN_TREE.gleed[d.size]):['ไม่โครง','รวมโครง'];
      h+='<label class="opt" style="width:100%;margin:8px 0 0;"><b style="color:#B3151D;">②</b> โครง '+rnChips(frg,d.frame,'frame')+'</label>';
    } else more=true;
  }
  if(more) h+=_rnStepHint();
  else h+='<div style="font-size:11px;color:#0369A1;margin-top:6px;">เลือก: '+(p.name||'').replace(/^บังตาผนังระแนง\s*/,'').replace(/^ระแนงผนัง\s*/,'')+' · เรต '+(p.rate||0).toLocaleString()+'/ตร.ม.</div>';
  return h+'</div>';
}
function rnDoorSwitch(ch){
  var ps=ch.querySelector('.i-prod'); var p=ps?PBYID[ps.value]:null; if(!p||!rnCatKey(p))return;
  var dt=ch.querySelector('.o-doortype'); var doortype=dt?dt.value:'';
  ch.dataset.rnDoorType=doortype;
  if(!doortype) return;
  var ck=rnCatKey(p), d=rnParse(p);
  if(d.frame==='ไม่โครง') return;
  var newId=(ck==='box')?rnResolve(ck,d.box,d.face,d.gap,'ไม่โครง'):rnResolve(ck,d.size,null,null,'ไม่โครง');
  if(newId && newId!==p.id) g3SyncProd(ch,newId);
}

BLOCK 2 — ลายระแนงประตูรั้ว (แทน 4 ฟังก์ชัน gateRnParse · gateRnResolve · gateRnChipsHTML · gateRnPick)

// ประตูรั้ว: ลายระแนง cascade ใหม่ (กล่อง→หน้าโชว์→ระยะ · ไม่โครงเท่านั้น) · ใช้ RN_TREE.box ร่วม
function gateRnParse(id){ var p=PBYID[id]; if(!p||rnCatKey(p)!=='box')return{box:RN_BOX_ORDER[0],face:'',gap:''}; var d=rnParse(p); return {box:d.box,face:d.face,gap:d.gap}; }
function gateRnResolve(box,face,gap){ var b=RN_TREE.box[box]; if(!b)return''; if(!b[face])face=Object.keys(b)[0]; var fc=b[face]; if(!fc)return''; if(!fc[gap])gap=Object.keys(fc)[0]; var node=fc[gap]; return (node&&node['ไม่โครง'])||''; }
function gateRnChipsHTML(curId){ var d=gateRnParse(curId);
  var b=RN_TREE.box[d.box]||{}; var faces=Object.keys(b); var fc=b[d.face]||{}; var gaps=Object.keys(fc);
  var rid=gateRnResolve(d.box,d.face,d.gap), rp=rid?PBYID[rid]:null;
  var h='<div class="gate-rn-casc" style="width:100%;border:1px solid #E5E7EB;border-radius:8px;padding:7px 9px;background:#FAFAFA;margin:4px 0;">';
  h+='<label class="opt" style="width:100%;margin:0;font-size:12px;">ลายระแนง · ขนาดกล่อง '+rnChips(RN_BOX_ORDER.filter(function(x){return RN_TREE.box[x];}),d.box,'box','gateRnPick')+'</label>';
  h+='<label class="opt" style="width:100%;margin:5px 0 0;font-size:12px;">หน้าโชว์ '+rnChips(faces,d.face,'face','gateRnPick')+'</label>';
  h+='<label class="opt" style="width:100%;margin:5px 0 0;font-size:12px;">ระยะห่าง '+rnChips(gaps,d.gap,'gap','gateRnPick')+'</label>';
  h+='<div style="font-size:11px;color:#0369A1;margin-top:5px;">'+(rp?(rp.name.replace(/^บังตาผนังระแนง\s*/,'').replace(/^ระแนงผนัง\s*/,'').replace(' ไม่โครง','')+' · เรต '+(rp.rate||0).toLocaleString()):'เลือกลาย')+'</div></div>';
  return h;
}
function gateRnPick(btn,dim){ var ch=btn.closest('.ch'); if(!ch)return; var sel=ch.querySelector('.o-gatern'); if(!sel)return;
  var cur=gateRnParse(sel.value), v=btn.dataset.val, box=cur.box, face=cur.face, gap=cur.gap;
  if(dim==='box'){ box=v; var b=RN_TREE.box[v]; face=Object.keys(b)[0]; gap=Object.keys(b[face])[0]; }
  else if(dim==='face'){ face=v; gap=Object.keys(RN_TREE.box[box][v])[0]; }
  else gap=v;
  var id=gateRnResolve(box,face,gap); if(!id)return;
  if(!sel.querySelector('option[value="'+id+'"]')){ var o=document.createElement('option'); o.value=id; sel.appendChild(o); }
  sel.value=id;
  var cont=ch.querySelector('.gate-rn-casc'); if(cont){ cont.outerHTML=gateRnChipsHTML(id); }
  calcQuote();
}
