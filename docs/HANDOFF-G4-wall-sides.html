  • ปรับขนาดต่อด้านได้ (ลิงก์ ✎ ซ่อน default · กรอก w/h → override auto)
  • ราคาผนังรวมเข้ายอดตู้ (_sell)

function cabWallToggle(chk, key){
  var ch = chk.closest('.ch');
  var det = ch.querySelector('.cab-wall-detail-' + key);
  if(det) det.style.display = chk.checked ? 'block' : 'none';
}

      optSel.cabwall_right_on  = true;  optSel.cabwall_right_mat = _oldMat;
      if(_bwOld && _bwOld.checked){
        optSel.cabwall_back_on   = true; optSel.cabwall_back_mat  = _oldMat;
      }

    // เก็บ legacy keys ไว้อ่าน (ไม่ลบ — กัน crash)
    const _cwmLeg=d.querySelector('.o-cabwallmat'); if(_cwmLeg)optSel.cabwallmat=_cwmLeg.value;
    const _bwLeg=d.querySelector('.o-backwall'); optSel.backwall=_bwLeg?_bwLeg.checked:false;
  }
