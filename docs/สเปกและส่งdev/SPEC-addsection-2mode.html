กลไก quick-add (non-breaking — จบที่การ์ดปกติ)

กดชิป = ทำ flow เพิ่มส่วนปกติ แต่ set กลุ่ม+แบบให้อัตโนมัติ (ข้ามกลุ่ม G1↔G3 ได้ · ยืนยันแล้ว):

function quickAddItem(group, prodId){
  addItem(document.getElementById('items'));          // ใช้ addItem เดิม
  const ch = document.querySelector('#items .ch:last-child');
  const grp = ch.querySelector('.i-group'); grp.value = group;
  grp.dispatchEvent(new Event('change',{bubbles:true}));
  const ps = ch.querySelector('.i-prod');
  ps.innerHTML = prodOptionsG6(String(group));         // rebuild ตามกลุ่ม
  ps.value = prodId;
  ps.dispatchEvent(new Event('change',{bubbles:true})); // → buildItemOpts/readItem ทำงานเหมือนเดิม
  ch.scrollIntoView({behavior:'smooth'});
}
ไม่พังของเดิม: quick-add แค่ "ลัดการเลือก" — ปลายทาง = การ์ดเดียวกับกดทั่วไป · ราคา/readItem/genQuote เหมือนเดิมเป๊ะ · ไม่แตะ logic
